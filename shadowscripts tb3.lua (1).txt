local configs = setmetatable({}, {__mode = "k"}) 
local originals = setmetatable({}, {__mode = "k"}) 
local lastCollect = 0 
local workerRunning = true 

local trackedKeys = { 
    "BulletColor", "BulletMaterial", "BulletTransparency", "BulletShape", "BulletSize", 
    "BulletLightEnabled", "BulletLightColor", "BulletMeshEnabled", "MuzzleFlashEnabled", 
    "MuzzleLightEnabled", "LightColor", "LightBrightness", "LightRange", "LightShadows", 
    "DisappearTime", "ShellScale", "BulletHoleEnabled", "BulletHoleTexture", 
    "BulletHoleVisibleTime", "BulletHoleSize", 
} 

local function saveOriginal(config) 
    if originals[config] then return end 
    local values = {} 
    for index, key in ipairs(trackedKeys) do 
        values[index] = rawget(config, key) 
    end 
    originals[config] = values 
end 

local function collectConfigs() 
    if type(getgc) ~= "function" then return end 
    local ok, objects = pcall(getgc, true) 
    if not ok or type(objects) ~= "table" then return end 
    for _, object in ipairs(objects) do 
        if typeof(object) == "table" then 
            local function add(config) 
                if typeof(config) == "table" and rawget(config, "BulletSpeed") then 
                    configs[config] = true 
                    saveOriginal(config) 
                end 
            end 
            pcall(function() 
                add(object) 
                add(rawget(object, 7)) 
                add(rawget(object, 8)) 
            end) 
        end 
    end 
    lastCollect = os.clock() 
end 

local function restoreConfig(config) 
    local saved = originals[config] 
    if not saved then return end 
    for index, key in ipairs(trackedKeys) do 
        pcall(function() rawset(config, key, saved[index]) end) 
    end 
end 

local function applyConfig(config) 
    if not config then return end 
    restoreConfig(config) 
    if state.Orb then 
        local color = Color3.fromHSV(state.Hue, 1, 1) 
        rawset(config, "BulletColor", color) 
        rawset(config, "BulletMaterial", Enum.Material.Neon) 
        rawset(config, "BulletTransparency", 0) 
        rawset(config, "BulletShape", Enum.PartType.Ball) 
        rawset(config, "BulletSize", Vector3.new(4, 4, 4)) 
        rawset(config, "BulletLightEnabled", true) 
        rawset(config, "BulletLightColor", color) 
        rawset(config, "MuzzleFlashEnabled", false) 
    end 
    if state.Flash then 
        local color = Color3.fromHSV((state.Hue + 0.18) % 1, 1, 1) 
        rawset(config, "MuzzleLightEnabled", true) 
        rawset(config, "LightColor", color) 
        rawset(config, "LightBrightness", 999) 
        rawset(config, "LightRange", 999) 
        rawset(config, "LightShadows", true) 
        rawset(config, "MuzzleFlashEnabled", true) 
        rawset(config, "DisappearTime", 8) 
    end 
    if state.Tornado then 
        rawset(config, "BulletShape", Enum.PartType.Ball) 
        rawset(config, "BulletMeshEnabled", false) 
    elseif state.Cylinder then 
        rawset(config, "BulletShape", Enum.PartType.Cylinder) 
        rawset(config, "BulletMeshEnabled", false) 
    end 
    if state.Shells then 
        rawset(config, "ShellScale", Vector3.new(state.ShellSize, state.ShellSize, state.ShellSize)) 
    end 
    if state.Spray then 
        rawset(config, "BulletHoleEnabled", true) 
        rawset(config, "BulletHoleTexture", {state.SprayTexture}) 
        rawset(config, "BulletHoleVisibleTime", state.SprayTime) 
        rawset(config, "BulletHoleSize", state.SpraySize) 
    end 
end 

local function anyEnabled() 
    return state.Orb or state.Flash or state.Cylinder or state.Tornado or state.Shells or state.Spray 
end 

local function applyAll() 
    if os.clock() - lastCollect >= 5 then 
        collectConfigs() 
    end 
    for config in pairs(configs) do 
        pcall(applyConfig, config) 
    end 
end 

local function restoreAll() 
    for config in pairs(configs) do 
        pcall(restoreConfig, config) 
    end 
end

    end 
end)

    local antiLoseEnabled = false local antiLoseCharConn = nil local antiLoseDiedConn = nil local savedToolNames = {}
    local function saveTools()
        savedToolNames = {} local backpack = player:FindFirstChild("Backpack") local character = player.Character
        if backpack then for _, tool in pairs(backpack:GetChildren()) do if tool:IsA("Tool") then table.insert(savedToolNames, tool.Name) end end end
        if character then for _, tool in pairs(character:GetChildren()) do if tool:IsA("Tool") then table.insert(savedToolNames, tool.Name) end end end
    end
    local function hookCharacter(char)
        local hum = char:WaitForChild("Humanoid", 5) if not hum then return end saveTools()

            local backpack = player:FindFirstChild("Backpack") if not backpack then return end

            if gunsFolder then
                for _, toolName in ipairs(savedToolNames) do
                    local model = gunsFolder:FindFirstChild(toolName)
                    local prompt = model and model:FindFirstChildWhichIsA("ProximityPrompt", true)
                    if prompt then pcall(function() prompt.HoldDuration = 0 fireproximityprompt(prompt) end) end
                end
            end
        end)
    end
    BypassGroupBox:Toggle({
        Name = "Anti Lose Items",
        Default = false,
        Callback = function(value)
            antiLoseEnabled = value
            if value then
                Library:Notification("Anti Lose Items enabled", "129793670226045", 3)
                if player.Character then task.spawn(function() hookCharacter(player.Character) end) end
                antiLoseCharConn = player.CharacterAdded:Connect(function(char) if antiLoseEnabled then task.spawn(function() hookCharacter(char) end) end end)

                if antiLoseCharConn then antiLoseCharConn:Disconnect() antiLoseCharConn = nil end
				 if antiLoseDiedConn then antiLoseDiedConn:Disconnect() antiLoseDiedConn = nil end savedToolNames = {}
            end

local function GetAttackEvent(tool) 
    if tool then

local function HasKnife(plr) 
    local character = plr.Character 
    local backpack = plr:FindFirstChild("Backpack") 
    if character then 
        for _, tool in pairs(character:GetChildren()) do 
            if tool:IsA("Tool") and allowedTools[tool.Name] then return true end 
        end 
    end 
    if backpack then 
        for _, tool in pairs(backpack:GetChildren()) do 
            if tool:IsA("Tool") and allowedTools[tool.Name] then return true end 
        end 
    end 
    return false 
end

    if not target or not target.Character then return end 
    
    local targetHRP = target.Character:FindFirstChild("HumanoidRootPart") 
    local humanoid = target.Character:FindFirstChild("Humanoid") 
    
    if not HasKnife(player) then 
        Library:Notification("Must buy knife first", "90449909165261", 5)
